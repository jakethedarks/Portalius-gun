local function create_portal(position, lookDirection)
	local portal = Instance.new("Part")
	portal.Name = "TeleportPortal"
	portal.Size = Vector3.new(6, 8, 0.25)
	portal.Transparency = 1
	portal.Anchored = true
	portal.CanCollide = false
	portal.CanTouch = true

	local direction = Vector3.new(lookDirection.X, 0, lookDirection.Z)
	if direction.Magnitude < 0.01 then direction = Vector3.new(0, 0, -1) else direction = direction.Unit end
	portal.CFrame = CFrame.lookAt(position + Vector3.new(0,4,0), position + Vector3.new(0,4,0) + direction)
	portal.Parent = workspace

	local decalFront = Instance.new("Decal")
	decalFront.Face = Enum.NormalId.Front
	decalFront.Texture = PORTAL_TEXTURE
	decalFront.Parent = portal

	local decalBack = Instance.new("Decal")
	decalBack.Face = Enum.NormalId.Back
	decalBack.Texture = PORTAL_TEXTURE
	decalBack.Parent = portal

	local light = Instance.new("PointLight")
	light.Color = Color3.fromRGB(70, 255, 0)
	light.Brightness = 7
	light.Range = 14
	light.Parent = portal

	local attachment = Instance.new("Attachment")
	attachment.Parent = portal

	local particles = Instance.new("ParticleEmitter")
	particles.Texture = "rbxasset://textures/particles/sparkles_main.dds"
	particles.Color = ColorSequence.new(Color3.fromRGB(80, 255, 0))
	particles.LightEmission = 1
	particles.Rate = 35
	particles.Lifetime = NumberRange.new(0.3, 0.8)
	particles.Speed = NumberRange.new(1, 4)
	particles.SpreadAngle = Vector2.new(360, 360)
	particles.Parent = attachment

	task.spawn(function()
		local originalSize = portal.Size
		while portal and portal.Parent do
			local scale = 1 + math.random(-4,4) / 100
			portal.Size = Vector3.new(originalSize.X * scale, originalSize.Y * scale, originalSize.Z)
			task.wait(0.06)
		end
	end)
	return portal
end

local function connectPortalTeleport(entrance, destination)
	if not entrance or not destination then return end
	entrance.Touched:Connect(function(hit)
		if teleportDebounce then return end
		local character = LocalPlayer.Character
		if not character or not hit:IsDescendantOf(character) then return end
		local hrp = character:FindFirstChild("HumanoidRootPart")
		if not hrp then return end

		teleportDebounce = true
		if portal_sound then portal_sound:Play() end
		hrp.CFrame = destination.CFrame * CFrame.new(0, 0, -3)
		task.wait(0.7)
		teleportDebounce = false
	end)
end

local function createPortalPair(posA, posB)
	clearAllPortals()
	portalA = create_portal(posA, posB - posA)
	portalB = create_portal(posB, posA - posB)
	connectPortalTeleport(portalA, portalB)
	connectPortalTeleport(portalB, portalA)
end

local function raycastFromScreen(screenPosition)
	local camera = workspace.CurrentCamera
	if not camera then return nil end
	local ray = camera:ViewportPointToRay(screenPosition.X, screenPosition.Y)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	if LocalPlayer.Character then params.FilterDescendantsInstances = {LocalPlayer.Character} end
	return workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
end

function setMode(mode)
	clearAllPortals()
	currentMode = mode
	if mode == "POINT" then
		updateStatus("POINT: DISPARA → TOCA A → TOCA B")
	elseif mode == "PLAYER" then
		updateStatus("PLAYER: INGRESA NOMBRE → DISPARA")
	elseif mode == "PLACE" then
		updateStatus("PLACE: INGRESA ID → DISPARA")
	end
end

local function startPointSelection()
	if currentMode ~= "POINT" then return end
	clearPointSelection()
	pointSelecting = true
	pointStage = 1
	updateStatus("TOCA LA PANTALLA → ESTABLECER PUNTO A")
end

local function setPoint(position)
	if currentMode ~= "POINT" or not pointSelecting then return end
	if pointStage == 1 then
		pointA = position
		pointStage = 2
		muzzleFlash()
		if portal_sound then portal_sound:Play() end
		updateStatus("PUNTO A ESTABLECIDO → TOCA LA PANTALLA PARA EL B")
		return
	end
	if pointStage == 2 then
		pointB = position
		muzzleFlash()
		if portal_sound then portal_sound:Play() end
		createPortalPair(pointA, pointB)
		pointStage = 0
		pointSelecting = false
		pointA = nil
		pointB = nil
		updateStatus("PORTALES LISTOS → ENTRA EN UNO")
	end
end

local function findPlayer(name)
	name = string.lower(string.gsub(name, "^%s*(.-)%s*$", "%1"))
	for _,player in ipairs(Players:GetPlayers()) do
		if string.lower(player.Name) == name or string.lower(player.DisplayName) == name then return player end
	end
	for _,player in ipairs(Players:GetPlayers()) do
		if string.lower(player.Name):sub(1, #name) == name then return player end
	end
	return nil
end

local function firePlayerPortal(playerName)
	local target = findPlayer(playerName)
	if not target or not target.Character then return false end
	local targetHRP = target.Character:FindFirstChild("HumanoidRootPart")
	local character = LocalPlayer.Character
	if not targetHRP or not character then return false end
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return false end

	muzzleFlash()
	if portal_sound then portal_sound:Play() end
	clearAllPortals()

	local entrance = create_portal(hrp.Position + hrp.CFrame.LookVector * 5, hrp.CFrame.LookVector)
	local destination = create_portal(targetHRP.Position, -targetHRP.CFrame.LookVector)
	portalA = entrance
	portalB = destination
	connectPortalTeleport(entrance, destination)
	return true
end

local function firePlacePortal(placeId)
	placeId = tonumber(placeId)
	if not placeId then return false end
	local character = LocalPlayer.Character
	if not character then return false end
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return false end

	muzzleFlash()
	if portal_sound then portal_sound:Play() end
	clearAllPortals()

	local portal = create_portal(hrp.Position + hrp.CFrame.LookVector * 5, hrp.CFrame.LookVector)
	portalA = portal
	portal.Touched:Connect(function(hit)
		if teleportDebounce then return end
		local char = LocalPlayer.Character
		if not char or not hit:IsDescendantOf(char) then return end
		teleportDebounce = true
		if portal_sound then portal_sound:Play() end
		local success = pcall(function() TeleportService:Teleport(placeId, LocalPlayer) end)
		if not success then teleportDebounce = false end
	end)
	return true
end

function fire()
	if currentMode == "PLAYER" then
		if targetInputText == "" then updateStatus("INGRESA UN NOMBRE DE JUGADOR") return end
		if firePlayerPortal(targetInputText) then updateStatus("ENTRA AL PORTAL") else updateStatus("JUGADOR NO ENCONTRADO") end
		return
	end
	if currentMode == "PLACE" then
		if targetInputText == "" then updateStatus("INGRESA UN ID DE JUEGO (PLACE ID)") return end
		if firePlacePortal(targetInputText) then updateStatus("ENTRA AL PORTAL") else updateStatus("ID DE JUEGO INVÁLIDO") end
		return
	end
	if currentMode == "POINT" then
		if not pointSelecting then startPointSelection() else updateStatus("TOCA LA PANTALLA PARA ESTABLECER EL PUNTO") end
	end
end

local function connectTool()
	if not portal_gun then return end
	portal_gun.Activated:Connect(function() fire() end)
end

-- CONTROLES MÓVILES (TOCAR LA PANTALLA) Y RATÓN PARA POINT

if UserInputService.TouchEnabled then
	UserInputService.TouchTap:Connect(function(touchPositions, processed)
		if processed or currentMode ~= "POINT" or not pointSelecting then return end
		local touch = touchPositions[1]
		if not touch then return end
		local result = raycastFromScreen(touch)
		if not result then updateStatus("SIN OBJETIVO VÁLIDO") return end
		setPoint(result.Position)
	end)
end

if UserInputService.MouseEnabled then
	UserInputService.InputBegan:Connect(function(inputObject, processed)
		if processed or inputObject.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
		if currentMode ~= "POINT" or not pointSelecting then return end
		local mousePosition = UserInputService:GetMouseLocation()
		local result = raycastFromScreen(mousePosition)
		if not result then updateStatus("SIN OBJETIVO VÁLIDO") return end
		setPoint(result.Position)
	end)
end
